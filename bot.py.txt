    async def start_command(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        try:
            user = update.effective_user
            db_user = self.db.query(User).filter_by(telegram_id=user.id).first()
            
            if not db_user:
                new_user = User(
                    telegram_id=user.id,
                    username=user.username,
                    first_name=user.first_name,
                    last_name=user.last_name,
                    join_date=datetime.utcnow()
                )
                self.db.add(new_user)
                streak = DailyStreak(user_id=user.id)
                self.db.add(streak)
                
                if context.args and context.args[0].isdigit():
                    referrer_id = int(context.args[0])
                    if referrer_id != user.id:
                        referrer = self.db.query(User).filter_by(telegram_id=referrer_id).first()
                        if referrer:
                            referrer.balance += self.config.REFERRAL_BONUS
                            referrer.total_earned += self.config.REFERRAL_BONUS
                            new_user.referred_by = referrer_id
                self.db.commit()
            
            await self.show_main_menu(update, context)
        except Exception as e:
            logger.error(f"Error in start_command: {e}")

    async def handle_callback(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        try:
            query = update.callback_query
            await query.answer()
            data = query.data
            user_id = update.effective_user.id
            user = self.db.query(User).filter_by(telegram_id=user_id).first()
            
            if data == 'profile':
                await self.show_profile(query, user)
            elif data == 'watch_ads':
                await self.show_ads(query, user)
            elif data == 'daily_tasks':
                await self.show_tasks(query, user)
            elif data == 'mini_games':
                await self.show_games(query, user)
            elif data == 'premium_menu':
                await self.show_premium(query, user)
            elif data == 'social_menu':
                await self.show_social_menu(query, user)
            elif data == 'payment_menu':
                await self.show_payment_menu(query, user)
            elif data == 'referral':
                await self.show_referral(query, user)
            elif data == 'back_to_menu':
                await self.show_main_menu_callback(query, user)
            elif data == 'admin_panel':
                await self.show_admin_panel(query, user)
            elif data.startswith('ad_watched_'):
                ad_id = data.split('_')[2]
                await self.ad_watched(query, user, ad_id)
            else:
                await query.edit_message_text("❌ Feature coming soon!")
        except Exception as e:
            logger.error(f"Error in handle_callback: {e}")

    async def show_ads(self, query, user):
        try:
            ads = self.db.query(Ad).filter_by(is_active=True).all()
            if not ads:
                await query.edit_message_text("❌ No ads available! Check back later.")
                return
            
            today = datetime.utcnow().replace(hour=0, minute=0, second=0)
            daily_views = self.db.query(AdView).filter(
                AdView.user_id == user.telegram_id,
                AdView.viewed_at >= today
            ).count()
            
            if daily_views >= self.config.MAX_ADS_PER_DAY:
                await query.edit_message_text(f"⚠️ You've watched {self.config.MAX_ADS_PER_DAY} ads today!\nCome back tomorrow.")
                return
            
            import random
            ad = random.choice(ads)
            remaining = self.config.MAX_ADS_PER_DAY - daily_views
            
            text = f"""
📺 **Watch Ad** ({remaining} remaining)

{ad.title}
{ad.description}

💰 Reward: **${ad.user_reward:.3f}**
👀 Views: {ad.total_views}/{ad.max_views}

[View Ad]({ad.link_url})
            """
            keyboard = [
                [InlineKeyboardButton("✅ I've Watched It", callback_data=f'ad_watched_{ad.id}')],
                [InlineKeyboardButton("🔙 Back", callback_data='back_to_menu')]
            ]
            await query.edit_message_text(text, reply_markup=InlineKeyboardMarkup(keyboard), parse_mode='Markdown')
        except Exception as e:
            logger.error(f"Error in show_ads: {e}")
    
    async def ad_watched(self, query, user, ad_id):
        try:
            ad = self.db.query(Ad).filter_by(id=int(ad_id)).first()
            if not ad:
                await query.edit_message_text("❌ Ad not found!")
                return
            
            today = datetime.utcnow().replace(hour=0, minute=0, second=0)
            daily_views = self.db.query(AdView).filter(
                AdView.user_id == user.telegram_id,
                AdView.viewed_at >= today
            ).count()
            
            if daily_views >= self.config.MAX_ADS_PER_DAY:
                await query.edit_message_text("⚠️ Daily limit reached!")
                return
            
            ad_view = AdView(
                user_id=user.telegram_id,
                ad_id=ad.id,
                viewed_at=datetime.utcnow(),
                user_reward=ad.user_reward,
                owner_revenue=ad.cost_per_view
            )
            self.db.add(ad_view)
            ad.total_views += 1
            if ad.total_views >= ad.max_views:
                ad.is_active = False
            
            user.balance += ad.user_reward
            user.total_earned += ad.user_reward
            self.db.commit()
            
            await query.edit_message_text(
                f"✅ **Ad Watched!**\n\n💰 You earned: ${ad.user_reward:.3f}\n💵 New Balance: ${user.balance:.2f}",
                reply_markup=InlineKeyboardMarkup([
                    [InlineKeyboardButton("📺 Watch More", callback_data='watch_ads')],
                    [InlineKeyboardButton("🔙 Back", callback_data='back_to_menu')]
                ]),
                parse_mode='Markdown'
            )
        except Exception as e:
            logger.error(f"Error in ad_watched: {e}")

    async def handle_message(self, update: Update, context: ContextTypes.DEFAULT_TYPE):
        try:
            user_id = update.effective_user.id
            text = update.message.text
            
            if text.lower().startswith('deposit'):
                parts = text.split()
                if len(parts) >= 2:
                    try:
                        amount = float(parts[1])
                        user = self.db.query(User).filter_by(telegram_id=user_id).first()
                        import uuid
                        deposit = Deposit(
                            user_id=user_id,
                            amount=amount,
                            method='mpesa',
                            transaction_id=f'DEP-{uuid.uuid4().hex[:8]}',
                            status='pending',
                            requested_at=datetime.utcnow()
                        )
                        self.db.add(deposit)
                        self.db.commit()
                        await update.message.reply_text(f"✅ Deposit of ${amount:.2f} recorded! Pending confirmation.")
                        return
                    except ValueError:
                        await update.message.reply_text("❌ Invalid amount!")
                        return
            
            if text.lower().startswith('withdraw'):
                parts = text.split()
                if len(parts) >= 3:
                    try:
                        amount = float(parts[1])
                        account = ' '.join(parts[2:])
                        user = self.db.query(User).filter_by(telegram_id=user_id).first()
                        
                        if amount < self.config.MIN_WITHDRAWAL:
                            await update.message.reply_text(f"❌ Min withdrawal is ${self.config.MIN_WITHDRAWAL:.2f}")
                            return
                        if amount > user.balance:
                            await update.message.reply_text(f"❌ Insufficient balance! You have ${user.balance:.2f}")
                            return
                        
                        import uuid
                        withdrawal = Withdrawal(
                            user_id=user_id,
                            amount=amount,
                            method='mpesa',
                            account_details=account,
                            transaction_id=f'WTH-{uuid.uuid4().hex[:8]}',
                            status='pending',
                            requested_at=datetime.utcnow()
                        )
                        self.db.add(withdrawal)
                        user.balance -= amount
                        self.db.commit()
                        await update.message.reply_text(f"✅ Withdrawal of ${amount:.2f} requested! Processing soon.")
                        return
                    except ValueError:
                        await update.message.reply_text("❌ Invalid amount!")
                        return
            
            await update.message.reply_text(
                "❓ I don't understand that.\n\n"
                "📱 **Available Commands:**\n"
                "/start - Main menu\n"
                "deposit [amount] - Deposit money\n"
                "withdraw [amount] [phone] - Withdraw money\n\n"
                "Or use the buttons below! 🔘"
            )
        except Exception as e:
            logger.error(f"Error in handle_message: {e}")
    
    def run(self):
        try:
            self.application = Application.builder().token(self.config.BOT_TOKEN).build()
            
            self.application.add_handler(CommandHandler("start", self.start_command))
            self.application.add_handler(CallbackQueryHandler(self.handle_callback))
            self.application.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, self.handle_message))
            
            logger.info(f"🚀 AdVantage Kenya Bot v{self.config.VERSION} is running...")
            self.application.run_polling()
        except Exception as e:
            logger.error(f"Failed to start bot: {e}")
            raise
